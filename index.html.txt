<script>
const metaData = { USD: { flag: '🇺🇸', name: '달러 / 원', pair: 'USD/KRW' }, JPY: { flag: '🇯🇵', name: '엔 / 원', pair: 'JPY/KRW (100엔)' }, EUR: { flag: '🇪🇺', name: '유로 / 원', pair: 'EUR/KRW' }, AUD: { flag: '🇦🇺', name: '호주달러 / 원', pair: 'AUD/KRW' }, THB: { flag: '🇹🇭', name: '바트 / 원', pair: 'THB/KRW' }};
const order = ['USD', 'JPY', 'EUR', 'AUD', 'THB'];
const subMap = {"RSI": "상대강도 (RSI14)", "MOM5": "단기 모멘텀 (MOM5)", "MOM20": "중기 추세 (MOM20)", "SYNC": "다통화 동조성 (SYNC)", "VOL": "변동성 비율 (VOL)", "DXY": "달러 이격도 (DXY)"};

let appMode = 'traveler', currentDDay = 14, globalData = {}, activeCurrency = null; 

async function initialize() { updateTime(); setInterval(updateTime, 1000); await fetchAllData(); }
async function fetchAllData() {
    document.getElementById('loading').style.display = 'flex';
    try {
        const promises = order.map(k => fetch(`/api/score?currency=${k}&d_day=${currentDDay}`).then(r => r.json()));
        const results = await Promise.all(promises);
        results.forEach((data, i) => { globalData[order[i]] = data; });
        renderMasterList(); if (activeCurrency) updateDetailView(activeCurrency);
    } catch (e) { console.error(e); }
    document.getElementById('loading').style.display = 'none';
}

function getGradeInfo(grade) {
    if(grade === 'A') return { text: "싸다", guide: "환전하세요!" };
    if(grade === 'B') return { text: "싼편", guide: "환전 고려" };
    if(grade === 'C') return { text: "보통", guide: "급하면 OK" };
    if(grade === 'D') return { text: "비싼편", guide: "기다려보세요" };
    return { text: "비싸다", guide: "지금은 비추" };
}

function renderMasterList() {
    document.getElementById('cardList').innerHTML = order.map(k => {
        const d = globalData[k], meta = metaData[k]; if(!d) return '';
        const info = getGradeInfo(d.grade);
        return `<div class="currency-card" onclick="openDetail('${k}')"><div class="card-left"><div class="flag-box">${meta.flag}</div><div><div class="cur-name">${meta.name}</div><div class="cur-pair">${meta.pair}</div></div></div><div class="card-right"><div class="cur-price">${d.price.toLocaleString('ko-KR', {minimumFractionDigits: 2, maximumFractionDigits: 2})}</div><div class="cur-grade c-${d.grade}">${d.rsi}점 · ${info.text}</div></div></div>`;
    }).join('');
}

function setMode(mode) {
    appMode = mode; document.getElementById('bodyTheme').className = mode;
    document.getElementById('btn-traveler').classList.toggle('active', mode === 'traveler');
    document.getElementById('btn-investor').classList.toggle('active', mode === 'investor');
    
    document.getElementById('traveler-gauge').style.display = mode === 'traveler' ? 'flex' : 'none';
    document.getElementById('quant-sub-panel').style.display = mode === 'traveler' ? 'block' : 'none'; 
    
    document.getElementById('dday-panel').style.display = mode === 'traveler' ? 'block' : 'none';
    document.getElementById('investor-board').style.display = mode === 'investor' ? 'block' : 'none';
    document.getElementById('macro-panel').style.display = mode === 'investor' ? 'block' : 'none';
    document.getElementById('traveler-action').style.display = mode === 'traveler' ? 'block' : 'none';
    document.getElementById('investor-action').style.display = mode === 'investor' ? 'block' : 'none';
    
    if(activeCurrency) updateDetailView(activeCurrency);
}

function togglePanel(contentId, headerElement) {
    const content = document.getElementById(contentId);
    if (content.style.display === 'none') {
        content.style.display = ''; 
        headerElement.classList.remove('collapsed');
    } else {
        content.style.display = 'none';
        headerElement.classList.add('collapsed');
    }
}

function openDetail(currencyCode) {
    activeCurrency = currencyCode;
    document.getElementById('list-view').style.transform = "translateX(-20%)"; document.getElementById('list-view').style.opacity = "0";
    setTimeout(() => { document.getElementById('list-view').style.display = 'none'; }, 300);
    const detailView = document.getElementById('detail-view'); detailView.style.display = 'block';
    setTimeout(() => { detailView.classList.add('active'); }, 10);
    updateDetailView(currencyCode);
}

function updateDetailView(currencyCode) {
    const d = globalData[currencyCode], meta = metaData[currencyCode];
    document.getElementById('d-flag').textContent = meta.flag; document.getElementById('d-name').textContent = meta.name;
    document.getElementById('d-pair').textContent = meta.pair; document.getElementById('d-price').textContent = d.price.toLocaleString('ko-KR', {minimumFractionDigits: 2, maximumFractionDigits: 2});
    
    document.getElementById('needleG').setAttribute('transform', `rotate(${-90 + (d.rsi / 100) * 180}, 150, 150)`);
    const gScore = document.getElementById('d-score'); gScore.textContent = d.rsi; gScore.className = `gauge-score c-${d.grade}`; gScore.style.background = 'transparent'; 
    const info = getGradeInfo(d.grade);
    document.getElementById('d-gradeText').textContent = `${d.grade}등급 (${info.text}·${info.guide})`;

    if (d.is_investable && d.inv_conds) {
        document.getElementById('inv-checklist-container').innerHTML = d.inv_conds.map(cond => {
            const statusClass = cond.pass ? 'pass' : 'fail';
            const statusColor = cond.pass ? '#1D9E75' : '#E24B4A';
            return `<div class="inv-check-item ${statusClass}"><div style="font-weight:700;">${cond.name}</div><div class="inv-val-box"><div class="inv-target">목표: ${cond.target}</div><div class="inv-curr" style="color:${statusColor}">${cond.current}</div></div></div>`;
        }).join('');
    } else {
        document.getElementById('inv-checklist-container').innerHTML = `<div style="text-align:center; padding: 20px; color:var(--text-sub); font-size:12px;">🚫 해당 통화는 수수료 및 성향 문제로 퀀트 투자를 지원하지 않습니다.</div>`;
    }

    document.getElementById('d-aiMsg').innerHTML = d.ai_message;
    
    const statsContainer = document.getElementById('stats-container');
    if (d.stat_details && Object.keys(d.stat_details).length > 0) {
        let statsHtml = `<div class="stat-box"><div class="stat-val">${d.stat_details.win_rate}%</div><div class="stat-label">과거 승률</div></div>`;
        statsHtml += `<div class="stat-box"><div class="stat-val">${d.stat_details.saving > 0 ? '+' : ''}${(d.stat_details.saving).toFixed(2)}%</div><div class="stat-label">평균 절감률</div></div>`;
        statsContainer.innerHTML = statsHtml;
        statsContainer.style.display = 'grid';
    } else {
        statsContainer.style.display = 'none';
    }
    
    if (d.sub_details) {
        document.getElementById('sub-scores-container').innerHTML = Object.entries(d.sub_details).map(([key, data]) => {
            const val = data.score;
            let color = "#888888"; 
            let gradeBadge = "";
            let statText = "";

            if (data.grade) {
                if (data.grade === 'A') color = "#1D9E75";
                else if (data.grade === 'B') color = "#5DCAA5";
                else if (data.grade === 'C') color = "#888888";
                else if (data.grade === 'D') color = "#EF9F27";
                else if (data.grade === 'E') color = "#E24B4A";
                
                gradeBadge = `<span style="display:inline-block; margin-left:6px; padding:2px 6px; border-radius:4px; background:${color}20; color:${color}; font-size:10px; font-weight:800;">${data.grade}</span>`;
                
                let advice = "";
                if (data.saving > 0.5) advice = " · <span style='color:#1D9E75; font-weight:700;'>유리한 구간!</span>";
                else if (data.saving > 0.1) advice = " · <span style='color:#5DCAA5;'>약간 유리</span>";
                else if (data.saving > -0.1) advice = " · <span style='color:#888888;'>보통</span>";
                else if (data.saving > -0.3) advice = " · <span style='color:#EF9F27;'>약간 불리</span>";
                else advice = " · <span style='color:#E24B4A; font-weight:700;'>불리한 구간</span>";

                statText = `<div class="sub-stat-text" style="margin-top:4px;">이 점수대에서 환전 시 승률 <strong>${data.win}%</strong> · 절감률 <strong>${data.saving > 0 ? '+' : ''}${data.saving.toFixed(2)}%</strong>${advice}</div>`;
            } else {
                color = val <= 40 ? "#1D9E75" : (val <= 60 ? "#888888" : "#E24B4A");
            }

            return `
            <div class="sub-row">
                <div class="sub-label">
                    <div style="display:flex; align-items:center;">
                        <span>${subMap[key] || key}</span>
                        ${gradeBadge}
                    </div>
                    <span style="color:${color}; font-family:'Space Mono', monospace; font-size:14px; font-weight:800;">${val}점</span>
                </div>
                <div class="progress-bg">
                    <div class="progress-fill" style="width:${val}%; background:${color}"></div>
                </div>
                ${statText}
            </div>`;
        }).join('');
    }

    document.getElementById('d-cutoffText').textContent = currentDDay <= 1 ? `🎯 D-1: 출국 임박 (무조건 환전)` : `🎯 D-${currentDDay} 목표 컷오프: ${d.threshold}점 이하`;
    const tBtn = document.getElementById('d-actionBtn');
    if (d.signal) { tBtn.textContent = "✅ 목표 컷오프 도달! (환전 권장)"; tBtn.style.background = "#1D9E75"; tBtn.style.color = "white"; }
    else { tBtn.textContent = "⏳ 컷오프 대기 중..."; tBtn.style.background = "var(--btn-bg)"; tBtn.style.color = "var(--btn-text)"; }

    const iBtn = document.getElementById('inv-actionBtn');
    if (!d.is_investable) {
        document.getElementById('i-cutoffText').textContent = "🚫 투자 부적합 통화";
        iBtn.textContent = "수수료 및 성향 문제 (접근 금지)";
        iBtn.style.background = "#2a2e37"; iBtn.style.color = "#a0a5b1"; iBtn.style.cursor = "not-allowed";
    } else {
        iBtn.style.cursor = "pointer";
        if (d.inv_signal) {
            document.getElementById('i-cutoffText').textContent = `🎯 매수 타점 포착! | ${d.inv_tpsl}`;
            iBtn.textContent = `🔥 기계적 매수 진입 (${d.inv_strategy})`;
            iBtn.style.background = "#E24B4A"; iBtn.style.color = "white";
        } else {
            document.getElementById('i-cutoffText').textContent = `⏳ 조건 미달 대기 중 | ${d.inv_tpsl}`;
            iBtn.textContent = "알고리즘 조건 미달 (관망)";
            iBtn.style.background = "var(--btn-bg)"; iBtn.style.color = "var(--btn-text)";
        }
    }

    if(d.macro_items && d.macro_items.length > 0) {
        const warnEmoji = d.macro_count === 0 ? '🟢' : d.macro_count <= 2 ? '🟡' : '🔴';
        const totalMacros = d.macro_items.length;
        document.getElementById('macro-header-text').innerHTML = `🌍 맞춤 경고 | ${warnEmoji} ${d.macro_count}/${totalMacros}개`;

        document.getElementById('macro-grid').innerHTML = d.macro_items.map(ind => `
            <div class="macro-item" style="border-left: 3px solid ${ind.w ? '#E24B4A' : '#1D9E75'}; display: flex; flex-direction: column; gap: 4px;">
                <div style="display: flex; justify-content: space-between; align-items: center;">
                    <span>${ind.w ? '⚠️' : '✅'} ${ind.n}</span>
                    <span style="color: ${ind.w ? '#E24B4A' : '#1D9E75'}; font-weight:700; font-family:'Space Mono', monospace;">
                        ${ind.v}
                    </span>
                </div>
                <div style="font-size: 10px; color: var(--text-sub); text-align: left; margin-left: 20px;">
                    ${ind.c}
                </div>
            </div>
        `).join('');

        // 🌟 EUR 등 특별 안내문 렌더링
        if (currencyCode === 'EUR') {
            document.getElementById('macro-disclaimer').innerHTML = `<div style="font-size: 11px; color: var(--text-sub); text-align: center; padding: 10px; background: rgba(255,255,255,0.05); border-radius: 8px;">ℹ️ EUR은 대부분 거시지표의 영향이 약합니다.<br>KOSPI만 강력한 경고 지표입니다.</div>`;
        } else {
            document.getElementById('macro-disclaimer').innerHTML = "";
        }
    } else { 
        document.getElementById('macro-header-text').innerHTML = `🌍 거시경제 꼬리위험`;
        document.getElementById('macro-grid').innerHTML = ""; 
        document.getElementById('macro-disclaimer').innerHTML = "";
    }
}
